int ngd_event_module_loop(int timeout)
{
    int n = epoll_wait(timeout);
    for (int i=0; i<n; i++)
    {
        conn_t *c = epoll_events[i].data.ptr;
        uint32_t evflag = epoll_events[i].flags;
        if (evflag & EPOLLIN)
            c->on_read = true;
        if (evflag & EPOLLOUT)
            c->on_write = true;
        c->handler(c);
    }
}
